% ===========================================================================
% Course math macros — target-neutral content semantics ONLY.
% (Anything about *rendering* — figures, circled labels, announcements —
% lives in the per-target preambles, not here.)
%
% Growth conventions:
%   - Add macros here the FIRST time a second lecture needs them; a macro used
%     by one lecture may start lecture-local in its content file.
%   - Promoting a macro is TWO edits, not one: add it here as \providecommand
%     AND delete the local definition from the lecture that had it.  This file
%     is loaded before any lecture body, so a surviving local \newcommand dies
%     with "Command \foo already defined" -- \providecommand here does NOT
%     rescue it.  Re-run the originating lecture's builds after every promotion.
%   - Record each promotion in CHANGELOG.md.
% ===========================================================================

% Number sets
\newcommand{\RR}{\mathbb{R}}
\newcommand{\ZZ}{\mathbb{Z}}

% Upright differential
\newcommand{\dd}{\mathrm{d}}

% Dirac notation
\providecommand{\ket}[1]{\left|#1\right\rangle}
\providecommand{\bra}[1]{\left\langle#1\right|}
\providecommand{\braket}[2]{\left\langle#1\,\middle|\,#2\right\rangle}

% Set-builder and definitional equality
\newcommand{\setof}[2]{\left\{\,#1\,\middle|\,#2\,\right\}}
\newcommand{\defeq}{\coloneqq}

% Upright space-group / little-group symbols.  Started lecture-local and was
% promoted here when a second lecture needed it -- the two-edit rule above:
% the originating lecture's local \newcommand was deleted in the same change.
\providecommand{\sg}[1]{\mathrm{#1}}
